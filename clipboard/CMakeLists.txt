cmake_minimum_required(VERSION 3.22)
if(ANDROID)
  add_library(clipboard STATIC android/clipboard.c)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(clipboard STATIC apple/clipboard.m)
  target_compile_options(clipboard PRIVATE -fobjc-arc)
  target_link_libraries(clipboard PUBLIC "-framework UIKit")
endif()
if(TARGET clipboard)
  target_include_directories(clipboard PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
  target_compile_features(clipboard PUBLIC c_std_99)
  set_target_properties(clipboard PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
if(TARGET clipboard)
  target_link_libraries(clipboard PUBLIC text_buffers)
endif()
if(ANDROID)
  # Java calls this entry point; retain it even before C consumers use the service.
  target_link_options(clipboard INTERFACE "-Wl,-u,Java_dev_drawdo_clipboard_Clipboard_install")
endif()
