cmake_minimum_required(VERSION 3.22)
add_library(accessible_ui STATIC access.c events.c)
target_include_directories(accessible_ui PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
target_compile_features(accessible_ui PUBLIC c_std_99)
target_link_libraries(accessible_ui PUBLIC text_buffers)
if(ANDROID)
  add_library(accessible_ui_native STATIC android/access.c)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(accessible_ui_native STATIC apple/access.m)
  target_compile_options(accessible_ui_native PRIVATE -fobjc-arc)
  target_link_libraries(accessible_ui_native PUBLIC "-framework UIKit")
endif()
if(TARGET accessible_ui_native)
  target_include_directories(accessible_ui_native PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
  target_compile_features(accessible_ui_native PUBLIC c_std_99)
  set_target_properties(accessible_ui_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
if(TARGET accessible_ui_native)
  target_link_libraries(accessible_ui_native PUBLIC accessible_ui)
endif()
set_target_properties(accessible_ui PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  # Java calls this entry point; retain it even before C consumers use the service.
  target_link_options(accessible_ui_native INTERFACE "-Wl,-u,Java_dev_drawdo_accessibleui_Accessibility_install")
endif()
