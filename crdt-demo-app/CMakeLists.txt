cmake_minimum_required(VERSION 3.22)
project(crdt_demo LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DATAFRIEND_TESTS OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_language(OBJC)
  set(DATAFRIEND_SHARED OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${REPO}/datafriend" datafriend)
add_subdirectory("${REPO}/text_crdt" text_crdt)
add_subdirectory("${REPO}/text_buffers" text_buffers)
add_library(demo_model STATIC src/document.c src/app.c)
target_link_libraries(demo_model PUBLIC datafriend::datafriend text_crdt text_buffers)
target_include_directories(demo_model PUBLIC src "${REPO}/native")
target_compile_options(demo_model PRIVATE -Wall -Wextra -Werror -pedantic-errors)
set_target_properties(demo_model PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  add_library(crdt_demo SHARED src/main.c src/sokol.c src/android.c "${REPO}/native/loop_android.c")
  target_compile_definitions(crdt_demo PRIVATE SOKOL_GLES3)
  target_link_libraries(crdt_demo PRIVATE demo_model text_buffers_native android log EGL GLESv3 m)
  target_link_options(crdt_demo PRIVATE -Wl,--no-undefined)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(DEVELOPMENT_TEAM "" CACHE STRING "Apple signing team")
  add_executable(crdt_demo MACOSX_BUNDLE src/main.c src/sokol.c src/apple.m "${REPO}/native/loop_apple.m")
  set_source_files_properties(src/sokol.c PROPERTIES LANGUAGE OBJC)
  set_source_files_properties(src/sokol.c src/apple.m PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_compile_definitions(crdt_demo PRIVATE SOKOL_METAL)
  target_link_libraries(crdt_demo PRIVATE demo_model text_buffers_native
    "-framework UIKit" "-framework Foundation" "-framework QuartzCore" "-framework Metal" "-framework MetalKit")
  set_target_properties(crdt_demo PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER dev.drawdo.crdt
    MACOSX_BUNDLE_BUNDLE_NAME "CRDT demo"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/ios/Info.plist.in"
    XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${DEVELOPMENT_TEAM}"
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER dev.drawdo.crdt
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
else()
  enable_testing()
  foreach(name document editor)
    add_executable(${name}_test tests/${name}_test.c)
    target_link_libraries(${name}_test PRIVATE demo_model)
    add_test(NAME ${name}_test COMMAND ${name}_test)
  endforeach()
endif()
option(CRDT_DEMO_EVENTS "Process network/editor events independently of rendering" ON)
if(TARGET crdt_demo)
  target_compile_definitions(crdt_demo PRIVATE CRDT_DEMO_EVENTS=$<BOOL:${CRDT_DEMO_EVENTS}>)
  target_include_directories(crdt_demo PRIVATE "${REPO}/third_party")
endif()

option(CRDT_DEMO_TIMING "Instrument native text latency in an isolated benchmark store" OFF)
if(CRDT_DEMO_TIMING)
  if(NOT TARGET text_buffers_native)
    message(FATAL_ERROR "Native timing requires an Android or iOS target")
  endif()
  target_sources(demo_model PRIVATE bench/timing.c bench/probe.c tests/native.c)
  target_include_directories(demo_model PUBLIC bench)
  target_compile_definitions(demo_model PUBLIC CRDT_DEMO_TIMING TEXT_BUFFERS_TRACE)
  target_compile_definitions(text_buffers_native PRIVATE TEXT_BUFFERS_TRACE)
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_sources(crdt_demo PRIVATE bench/capture.m)
    set_source_files_properties(bench/capture.m PROPERTIES COMPILE_OPTIONS -fobjc-arc)
    target_link_libraries(crdt_demo PRIVATE "-framework ReplayKit" "-framework CoreImage"
      "-framework CoreMedia" "-framework CoreVideo" "-framework ImageIO")
  endif()
endif()
