cmake_minimum_required(VERSION 3.22)
add_library(text_buffers STATIC text.c editor.c)
target_include_directories(text_buffers PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
target_compile_features(text_buffers PUBLIC c_std_99)
if(ANDROID)
  add_library(text_buffers_native STATIC android/editor.c)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(text_buffers_native STATIC apple/editor.m)
  target_compile_options(text_buffers_native PRIVATE -fobjc-arc)
  target_link_libraries(text_buffers_native PUBLIC "-framework UIKit" "-framework CoreGraphics")
endif()
if(TARGET text_buffers_native)
  target_include_directories(text_buffers_native PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
  target_compile_features(text_buffers_native PUBLIC c_std_99)
  set_target_properties(text_buffers_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
if(TARGET text_buffers_native)
  target_link_libraries(text_buffers_native PUBLIC text_buffers)
endif()
set_target_properties(text_buffers PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  # Java calls this entry point; retain it even before C consumers use the service.
  target_link_options(text_buffers_native INTERFACE "-Wl,-u,Java_dev_drawdo_textbuffers_TextEditor_install")
endif()
