cmake_minimum_required(VERSION 3.22)
if(ANDROID)
  add_library(haptics STATIC android/haptics.c)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(haptics STATIC apple/haptics.m)
  target_compile_options(haptics PRIVATE -fobjc-arc)
  target_link_libraries(haptics PUBLIC "-framework UIKit")
endif()
if(TARGET haptics)
  target_include_directories(haptics PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
  target_compile_features(haptics PUBLIC c_std_99)
  set_target_properties(haptics PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
if(ANDROID)
  # Java calls this entry point; retain it even before C consumers use the service.
  target_link_options(haptics INTERFACE "-Wl,-u,Java_dev_drawdo_haptics_Haptics_install")
endif()
