cmake_minimum_required(VERSION 3.22)
project(drawdo LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DATAFRIEND_TESTS OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_language(OBJC)
  set(CMAKE_OBJC_STANDARD 99)
  set(CMAKE_OBJC_STANDARD_REQUIRED ON)
  set(DATAFRIEND_SHARED OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${REPO}/datafriend" datafriend)
add_subdirectory("${REPO}/text_buffers" text_buffers)
add_library(todo STATIC src/model.c)
target_include_directories(todo PUBLIC src "${REPO}" "${REPO}/native")
target_link_libraries(todo PUBLIC datafriend::datafriend)
target_compile_options(todo PRIVATE -Wall -Wextra -Werror -pedantic-errors)
set_target_properties(todo PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(APP_SOURCES src/main.c src/screen.c src/sokol.c src/render/text.c src/render/text_cache.c)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_subdirectory("${REPO}/accessible_ui" accessible_ui)
endif()
if(ANDROID)
  add_library(drawdo SHARED ${APP_SOURCES} src/android/platform.c src/android/text.c src/android/present.c "${REPO}/native/loop_android.c")
  target_compile_definitions(drawdo PRIVATE SOKOL_GLES3)
  target_link_libraries(drawdo PRIVATE todo text_buffers_native accessible_ui_native android log EGL GLESv3 jnigraphics m)
  target_link_options(drawdo PRIVATE -Wl,--no-undefined)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(DEVELOPMENT_TEAM "" CACHE STRING "Apple signing team")
  add_executable(drawdo MACOSX_BUNDLE ${APP_SOURCES} src/apple/platform.m src/apple/text.m src/apple/present.m "${REPO}/native/loop_apple.m")
  set_source_files_properties(src/sokol.c PROPERTIES LANGUAGE OBJC)
  set_source_files_properties(src/sokol.c src/apple/platform.m src/apple/text.m src/apple/present.m "${REPO}/native/loop_apple.m" PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_compile_definitions(drawdo PRIVATE SOKOL_METAL)
  target_link_libraries(drawdo PRIVATE todo text_buffers_native accessible_ui_native
    "-framework UIKit" "-framework Foundation" "-framework QuartzCore" "-framework CoreGraphics" "-framework Metal" "-framework MetalKit")
  set_target_properties(drawdo PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER dev.drawdo.app MACOSX_BUNDLE_BUNDLE_NAME DrawDo
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/ios/Info.plist.in"
    XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${DEVELOPMENT_TEAM}"
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER dev.drawdo.app
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
else()
  enable_testing()
  add_executable(model_test tests/model_test.c)
  target_link_libraries(model_test PRIVATE todo)
  add_test(NAME model COMMAND model_test)
  add_executable(network_test tests/network_test.c)
  target_link_libraries(network_test PRIVATE todo)
endif()
if(TARGET drawdo)
  target_include_directories(drawdo PRIVATE "${REPO}/third_party")
  target_compile_options(drawdo PRIVATE -Wall -Wextra)
endif()
