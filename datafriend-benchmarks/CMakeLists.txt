cmake_minimum_required(VERSION 3.18)
project(datafriend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(datafriend SHARED native/engine.cpp native/c_api.cpp native/aelmdb.c native/aelmdb_midl.c)
target_sources(datafriend PRIVATE native/identity.cpp native/identity_api.cpp
  native/identity_apple.cpp native/identity_android.cpp
  ../third_party/monocypher/monocypher.c ../third_party/monocypher/monocypher-ed25519.c)
target_include_directories(datafriend PRIVATE ../third_party/monocypher)
if(APPLE)
  target_link_libraries(datafriend PRIVATE "-framework Security" "-framework CoreFoundation")
endif()
target_include_directories(datafriend PUBLIC include ../third_party/unordered_dense/include PRIVATE ../third_party/aelmdb/src ../littlebear-benchmarks/src)
target_compile_definitions(datafriend PRIVATE MDB_HASH_SIZE=8)
target_link_libraries(datafriend PRIVATE Threads::Threads)
target_compile_options(datafriend PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
option(DATAFRIEND_DIRECT_SUBSCRIPTIONS "Experiment: concrete EA dependencies plus predicate match subscriptions" OFF)
if(DATAFRIEND_DIRECT_SUBSCRIPTIONS)
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_DIRECT_SUBSCRIPTIONS=1)
endif()
option(DATAFRIEND_BATCH_ROUTING "Experiment: route the complete pending batch before updating indexes" OFF)
if(DATAFRIEND_BATCH_ROUTING)
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_BATCH_ROUTING=1)
endif()
option(DATAFRIEND_BUFFER_QUERIES "Diagnostic only: buffer query facts before maintaining views" OFF)
if(DATAFRIEND_BUFFER_QUERIES)
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_BUFFER_QUERIES=1)
endif()
option(DATAFRIEND_DETAIL_VIEWS "Diagnostic only: separate view maintenance into measured passes" OFF)
if(DATAFRIEND_DETAIL_VIEWS)
  if(NOT DATAFRIEND_BUFFER_QUERIES)
    message(FATAL_ERROR "DATAFRIEND_DETAIL_VIEWS requires DATAFRIEND_BUFFER_QUERIES")
  endif()
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_DETAIL_VIEWS=1)
endif()
option(DATAFRIEND_PROFILE "Instrument native edits for phase timing" OFF)
if(DATAFRIEND_PROFILE)
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_PROFILE=1)
  add_executable(engine_profile benchmarks/profile.cpp)
  target_link_libraries(engine_profile PRIVATE datafriend)
endif()
option(DATAFRIEND_PROFILE_INDEXES "Diagnostic only: per-record index component clocks" OFF)
if(DATAFRIEND_PROFILE_INDEXES)
  if(NOT DATAFRIEND_PROFILE)
    message(FATAL_ERROR "DATAFRIEND_PROFILE_INDEXES requires DATAFRIEND_PROFILE")
  endif()
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_PROFILE_INDEXES=1)
  if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
    add_executable(index_clock_probe benchmarks/index_clock_probe.cpp)
  endif()
endif()
option(DATAFRIEND_SQLITE_BENCHMARK "Build the optional host SQLite comparison" OFF)
if(DATAFRIEND_SQLITE_BENCHMARK AND NOT ANDROID AND NOT IOS)
  find_package(SQLite3 REQUIRED)
  add_executable(sqlite_comparison benchmarks/sqlite.cpp)
  target_link_libraries(sqlite_comparison PRIVATE datafriend SQLite::SQLite3 Threads::Threads)
  target_include_directories(sqlite_comparison PRIVATE ../third_party/unordered_dense/include ../littlebear-benchmarks/src)
  target_compile_options(sqlite_comparison PRIVATE -Wall -Wextra)
endif()
option(DATAFRIEND_TESTS "Build native tests and benchmark" ON)
if(DATAFRIEND_TESTS AND NOT ANDROID AND NOT IOS)
  enable_testing()
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_TESTING=1)
  foreach(name engine_test persistent_tree_test async_flush_test sorted_dense_test network_test identity_test)
    add_executable(${name} tests/${name}.cpp)
    target_link_libraries(${name} PRIVATE datafriend)
    target_include_directories(${name} PRIVATE ../third_party/aelmdb/src ../littlebear-benchmarks/src ../third_party/monocypher)
    target_compile_definitions(${name} PRIVATE MDB_HASH_SIZE=8)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
  add_executable(engine_benchmark benchmarks/engine.cpp)
  target_link_libraries(engine_benchmark PRIVATE datafriend)
  add_executable(routing_benchmark benchmarks/fixed_routing.cpp)
  target_link_libraries(routing_benchmark PRIVATE datafriend)
endif()

option(DATAFRIEND_PORTABLE_BENCHMARKS "Build simulation and network checks for host/Android" OFF)
if(DATAFRIEND_PORTABLE_BENCHMARKS)
  add_executable(simulation_benchmark benchmarks/simulation.cpp)
  target_link_libraries(simulation_benchmark PRIVATE datafriend)
  if(NOT TARGET network_test)
    add_executable(network_test tests/network_test.cpp)
    target_link_libraries(network_test PRIVATE datafriend)
    target_include_directories(network_test PRIVATE ../littlebear-benchmarks/src)
  endif()
endif()
